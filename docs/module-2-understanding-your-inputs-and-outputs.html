def list_files_tree(data_dir_path, depth=0, bold_highlight=False, list_to_bold_highlight=None):

if list_to_bold_highlight is None:
    list_to_bold_highlight = []

#print only once
if depth == 0:
    print(f'TREE STRUCTURE OF OUTPUT DIRECTORY FOR {os.path.basename(data_dir_path)}')

ELBOW, SPACE = '|_', '   '
print(SPACE * depth, ELBOW , os.path.basename(data_dir_path))
if os.path.isdir(data_dir_path):
    contents = os.listdir(data_dir_path)
    for content in contents:
        if os.path.isdir(os.path.join(data_dir_path, content)):
            list_files_tree(os.path.join(data_dir_path, content), depth=depth+1,
                            bold_highlight=bold_highlight,
                            list_to_bold_highlight=list_to_bold_highlight)
        else:
            if bold_highlight and content in list_to_bold_highlight:
                print(SPACE * (depth+1), ELBOW , "\033[1;100m", content, "\033[0m")
            else:
                print(SPACE * (depth+1), ELBOW , content)

2. Self: CosMx (imaging / single-molecule)

Same idea as Xenium, different filenames. Nanostring / Bruker CosMx, human hippocampus WTX.

Useful files:

  • S0_tx_file.csv.gz — transcripts (molecule table). Listed in md5sums.txt; this copy of the folder may not include it
  • S0_metadata_file.csv.gz — per-cell metadata (centroids, area, RNA counts)
  • S0_exprMat_file.csv.gz — dense cell x gene matrix
  • S0-polygons.csv.gz — cell outlines
  • S0_fov_positions_file.csv.gz — FOV anchors on the slide

list_files = [‘S0_tx_file.csv.gz’, ‘S0_metadata_file.csv.gz’, ‘S0_exprMat_file.csv.gz’, ‘S0-polygons.csv.gz’, ‘S0_fov_positions_file.csv.gz’] list_files_tree(os.path.join(DATA_DIR_PATH, ‘cosmx’), depth=0, bold_highlight=True, list_to_bold_highlight=list_files)
